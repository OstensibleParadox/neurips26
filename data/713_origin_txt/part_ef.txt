## 方法2：直接通过Python下载最新安装包

```python
import requests
import os
import subprocess
from pathlib import Path

def update_firefox():
    # 获取最新Firefox下载链接（Windows版）
    # 实际使用时需要从Mozilla API获取最新链接
    firefox_url = "https://download.mozilla.org/?product=firefox-latest&os=win&lang=en-US"
    
    # 下载安装程序
    print("正在下载Firefox...")
    response = requests.get(firefox_url, stream=True)
    
    installer_path = Path("C:/Temp/FirefoxSetup.exe")
    installer_path.parent.mkdir(parents=True, exist_ok=True)
    
    with open(installer_path, 'wb') as f:
        for chunk in response.iter_content(chunk_size=8192):
            f.write(chunk)
    
    print("正在安装Firefox...")
    # 静默安装参数
    subprocess.run([str(installer_path), "/S"])
    
    # 清理安装文件
    installer_path.unlink()
    
    print("Firefox更新完成！")

if __name__ == "__main__":
    update_firefox()
```

## 方法4：完整的自动化脚本

```python
import wmi
import psutil
import requests
import subprocess
import tempfile
import os
from pathlib import Path

def close_firefox():
    """关闭所有Firefox进程"""
    for proc in psutil.process_iter(['name']):
        if proc.info['name'] and 'firefox' in proc.info['name'].lower():
            proc.terminate()
            proc.wait()

def get_firefox_version():
    """获取当前Firefox版本"""
    try:
        # Firefox通常安装在Program Files或Program Files (x86)
        possible_paths = [
            Path("C:/Program Files/Mozilla Firefox/firefox.exe"),
            Path("C:/Program Files (x86)/Mozilla Firefox/firefox.exe")
        ]
        
        for path in possible_paths:
            if path.exists():
                result = subprocess.run([str(path), "--version"], 
                                      capture_output=True, text=True)
                if result.returncode == 0:
                    return result.stdout.strip()
    except:
        pass
    return None

def update_firefox_complete():
    """完整的Firefox更新流程"""
    print("正在检查Firefox更新...")
    
    # 1. 关闭Firefox
    print("关闭Firefox进程...")
    close_firefox()
    
    # 2. 下载最新安装包
    print("下载最新版Firefox...")
    
    # 最新的稳定版下载链接
    firefox_url = "https://download.mozilla.org/?product=firefox-latest-ssl&os=win64&lang=en-US"
    
    with tempfile.TemporaryDirectory() as tmpdir:
        installer_path = Path(tmpdir) / "FirefoxInstaller.exe"
        
        # 下载
        response = requests.get(firefox_url, stream=True)
        response.raise_for_status()
        
        with open(installer_path, 'wb') as f:
            for chunk in response.iter_content(chunk_size=8192):
                f.write(chunk)
        
        # 3. 静默安装
        print("正在安装更新...")
        # /S = 静默安装，/MaintenanceService = 更新维护服务
        result = subprocess.run([str(installer_path), "/S", "/MaintenanceService"], 
                              capture_output=True, text=True)
        
        if result.returncode == 0:
            print("✅ Firefox更新成功！")
            new_version = get_firefox_version()
            if new_version:
                print(f"当前版本: {new_version}")
        else:
            print("❌ 更新失败")
            print(f"错误信息: {result.stderr}")

if __name__ == "__main__":
    update_firefox_complete()
```

## 方法1：使用系统注册表（Windows）

```python
import winreg
import subprocess
import json

def get_installed_software_registry():
    """通过注册表获取已安装软件列表"""
    software_list = []
    
    # 注册表路径（64位和32位软件）
    registry_paths = [
        r"SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall",
        r"SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall"
    ]
    
    for reg_path in registry_paths:
        try:
            key = winreg.OpenKey(winreg.HKEY_LOCAL_MACHINE, reg_path)
            for i in range(0, winreg.QueryInfoKey(key)[0]):
                try:
                    subkey_name = winreg.EnumKey(key, i)
                    subkey = winreg.OpenKey(key, subkey_name)
                    
                    software_name = None
                    software_version = None
                    software_publisher = None
                    
                    try:
                        software_name = winreg.QueryValueEx(subkey, "DisplayName")[0]
                        software_version = winreg.QueryValueEx(subkey, "DisplayVersion")[0]
                        software_publisher = winreg.QueryValueEx(subkey, "Publisher")[0]
                    except (FileNotFoundError, OSError):
                        continue
                    
                    if software_name:
                        software_list.append({
                            "name": software_name,
                            "version": software_version or "未知版本",
                            "publisher": software_publisher or "未知发布者",
                            "registry_path": f"{reg_path}\\{subkey_name}"
                        })
                    
                    winreg.CloseKey(subkey)
                except (OSError, WindowsError):
                    continue
            winreg.CloseKey(key)
        except (OSError, WindowsError):
            continue
    
    return software_list

# 使用示例
softwares = get_installed_software_registry()
print(f"找到 {len(softwares)} 个已安装软件")
for idx, software in enumerate(softwares[:10], 1):  # 显示前10个
    print(f"{idx}. {software['name']} - 版本: {software['version']}")
```

```python
def get_installed_software_wmi():
    """使用WMI获取已安装软件信息"""
```

```python
import subprocess
import platform
import re

def get_software_by_command():
    """使用系统命令获取软件信息"""
    system = platform.system()
    software_list = []
    
    if system == "Windows":
        # 使用PowerShell获取软件列表
        powershell_cmd = [
            "powershell",
            "Get-ItemProperty HKLM:\\Software\\Wow6432Node\\Microsoft\\Windows\\CurrentVersion\\Uninstall\\*, "
            "HKLM:\\Software\\Microsoft\\Windows\\CurrentVersion\\Uninstall\\*, "
            "HKCU:\\Software\\Microsoft\\Windows\\CurrentVersion\\Uninstall\\* | "
            "Select-Object DisplayName, DisplayVersion, Publisher, InstallDate | "
            "ConvertTo-Json"
        ]
        
        try:
            result = subprocess.run(powershell_cmd, 
                                  capture_output=True, 
                                  text=True, 
                                  encoding='utf-8',
                                  errors='ignore')
            
            if result.returncode == 0:
                import json
                data = json.loads(result.stdout)
                for item in data:
                    if item.get("DisplayName"):
                        software_list.append({
                            "name": item["DisplayName"],
                            "version": item.get("DisplayVersion", "未知版本"),
                            "publisher": item.get("Publisher", "未知发布者"),
                            "install_date": item.get("InstallDate", "")
                        })
        except:
            pass
    
    elif system == "Linux":
        # 使用dpkg（Debian/Ubuntu）
        try:
            result = subprocess.run(["dpkg", "-l"], 
                                  capture_output=True, 
                                  text=True)
            lines = result.stdout.split('\n')[5:]  # 跳过标题行
            
            for line in lines:
                if line.strip():
                    parts = line.split()
                    if len(parts) >= 3:
                        software_list.append({
                            "name": parts[1],
                            "version": parts[2],
                            "description": ' '.join(parts[3:]) if len(parts) > 3 else ""
                        })
        except:
            pass
        
        # 或者使用rpm（RedHat/CentOS）
        try:
            result = subprocess.run(["rpm", "-qa", "--queryformat", 
                                   '"%{NAME} %{VERSION}-%{RELEASE}\n"'], 
                                  capture_output=True, 
                                  text=True)
            for line in result.stdout.strip().split('\n'):
                if line:
                    name, version = line.strip('"').split(' ', 1)
                    software_list.append({
                        "name": name,
                        "version": version
                    })
        except:
            pass
    
    elif system == "Darwin":  # macOS
        try:
            result = subprocess.run(["system_profiler", "SPApplicationsDataType", "-json"], 
                                  capture_output=True, 
                                  text=True)
            import json
            data = json.loads(result.stdout)
            
            for app in data.get("SPApplicationsDataType", []):
                if "version" in app:
                    software_list.append({
                        "name": app.get("_name", "未知"),
                        "version": app.get("version", "未知版本"),
                        "path": app.get("path", "")
                    })
        except:
            pass
    
    return software_list
```
