def analyze_file(filepath, node_limit=None, similarity_threshold=0.8):
    """分析文件"""
    print(f"\n{'='*70}")
    print(f"🔍 NPS-4 NARRATIVE PATHOLOGY SCANNER")
    print(f"{'='*70}")
    print(f"📂 Analyzing: {filepath}")
    print(f"{'='*70}\n")
    
    # 读取文件
    try:
        with open(filepath, 'r', encoding='utf-8', errors='replace') as f:
            text = f.read()
        
        print(f"📄 File loaded: {len(text):,} characters")
        
        # 显示前几行预览
        lines = text.split('\n')
        non_empty_lines = [line for line in lines if line.strip()]
        print(f"📝 Preview (first 3 non-empty lines):")
        for i, line in enumerate(non_empty_lines[:3]):
            line_preview = line[:80] + "..." if len(line) > 80 else line
            print(f"   {i+1}: {line_preview}")
            
    except Exception as e:
        print(f"❌ Error reading file: {e}")
        return None
    
    # 构建叙事图
    builder = NarrativeGraphBuilder(
        similarity_threshold=similarity_threshold, 
        node_limit=node_limit
    )
    
    G = builder.build(text)
    
    if len(G.nodes()) == 0:
        print("❌ No nodes created. Analysis failed.")
        return None
    
    # 计算指标
    metrics_calculator = PathologyMetrics()
    metrics = metrics_calculator.calculate_all(G)
    
    # 创建可视化
    filename = os.path.basename(filepath)
    fig = create_visualization(G, metrics, filename)
    
    # 保存可视化
    out_path = None
    if fig:
        try:
            out_name = os.path.splitext(filename)[0]
            out_path = f"{OUTPUT_DIR}/{out_name}_analysis.png"
            
            fig.savefig(out_path, dpi=150, bbox_inches='tight', 
                       facecolor='white', edgecolor='none')
            plt.close(fig)
            print(f"\n💾 Visualization saved to: {out_path}")
        except Exception as e:
            print(f"⚠️ Error saving visualization: {e}")
    
    # 保存图形数据
    graph_data_path = None
    try:
        out_name = os.path.splitext(filename)[0]
        graph_data_path = f"{OUTPUT_DIR}/{out_name}_graph_data.json"
        
        # 准备图形数据
        graph_data = {
            "filename": filename,
            "analysis_time": datetime.now().isoformat(),
            "parameters": {
                "similarity_threshold": similarity_threshold,
                "node_limit": node_limit
            },
            "metrics": metrics,
            "top_nodes": sorted(
                [(node, G.degree(node)) for node in G.nodes()],
                key=lambda x: x[1],
                reverse=True
            )[:10],
            "node_samples": [
                {
                    "id": node,
                    "label": G.nodes[node].get('label', ''),
                    "text_preview": G.nodes[node].get('text', '')[:100],
                    "degree": G.degree(node)
                }
                for node in list(G.nodes())[:20]
            ]
        }
        
        # 保存JSON文件
        with open(graph_data_path, 'w', encoding='utf-8') as f:
            json.dump(graph_data, f, ensure_ascii=False, indent=2)
            
        print(f"💾 Graph data saved to: {graph_data_path}")
    except Exception as e:
        print(f"⚠️ Error saving graph data: {e}")
    
    # 显示结果
    print_results(metrics, filename, out_path, graph_data_path)
    
    return metrics
