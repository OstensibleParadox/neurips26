## 解决方案3：完整修复示例

这是一个修复后的完整示例文件：

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
更新Firefox浏览器脚本
适用于统信UOS系统
"""

import subprocess
import os
import sys
from pathlib import Path

def check_firefox_version():
    """检查当前Firefox版本"""
    print("正在检查Firefox版本...")
    try:
        result = subprocess.run(["firefox", "--version"], 
                              capture_output=True, 
                              text=True)
        if result.returncode == 0:
            version = result.stdout.strip()
            print(f"当前版本: {version}")
            return version
    except:
        print("无法获取Firefox版本")
    return None

def update_firefox_uos():
    """在UOS上更新Firefox"""
    print("=" * 50)
    print("Firefox更新工具")
    print("=" * 50)
    
    # 检查当前版本
    current_version = check_firefox_version()
    
    # 更新Firefox
    print("\n正在更新Firefox...")
    
    try:
        # 方法1: 使用apt更新
        print("方法1: 使用apt更新...")
        commands = [
            ["sudo", "apt", "update"],
            ["sudo", "apt", "install", "--only-upgrade", "firefox", "-y"],
            ["sudo", "apt", "autoremove", "-y"],
            ["sudo", "apt", "clean"]
        ]
        
        for cmd in commands:
            print(f"执行: {' '.join(cmd)}")
            result = subprocess.run(cmd, text=True)
            if result.returncode != 0:
                print(f"命令执行失败: {result.returncode}")
        
        # 方法2: 如果apt失败，尝试其他方法
        print("\n检查更新结果...")
        new_version = check_firefox_version()
        
        if new_version:
            if current_version != new_version:
                print(f"✓ 更新成功!")
                print(f"  旧版本: {current_version}")
                print(f"  新版本: {new_version}")
            else:
                print("✓ Firefox已经是最新版本")
        else:
            print("✗ 无法验证更新是否成功")
            
    except Exception as e:
        print(f"更新过程中出错: {e}")
        print("请尝试手动更新:")
        print("  1. 打开终端 (Ctrl+Alt+T)")
        print("  2. 运行: sudo apt update && sudo apt upgrade firefox")

def main():
    """主函数"""
    # 检查是否为UOS/Linux系统
    if sys.platform != "linux":
        print("此脚本仅适用于Linux系统（如统信UOS）")
        return
    
    # 检查是否有sudo权限
    if os.geteuid() != 0:
        print("注意: 需要管理员权限更新系统软件")
        print("请输入密码继续...")
    
    # 执行更新
    update_firefox_uos()
    
    print("\n" + "=" * 50)
    print("更新完成！")
    print("=" * 50)

if __name__ == "__main__":
    main()
```

### 方案4：兼容性更好的完整脚本

这里是一个完全兼容Python 3.0+的版本：

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
统信UOS Firefox更新脚本
兼容Python 3.0+ 版本
"""

import subprocess
import os
import sys
import platform

def run_command(cmd, show_output=True):
    """运行命令并返回结果"""
    if show_output:
        print("[执行] " + " ".join(cmd))
    
    try:
        result = subprocess.run(cmd, 
                              capture_output=True, 
                              text=True,
                              timeout=30)
        if show_output and result.stdout:
            print(result.stdout)
        if result.stderr:
            print("错误: " + result.stderr)
        return result.returncode, result.stdout, result.stderr
    except subprocess.TimeoutExpired:
        print("命令执行超时")
        return -1, "", "Timeout"
    except Exception as e:
        print("执行错误: " + str(e))
        return -1, "", str(e)

def check_system():
    """检查系统类型"""
    system = platform.system()
    print("系统类型: " + system)
    
    if system != "Linux":
        print("错误: 此脚本仅适用于Linux系统")
        return False
    
    # 检查是否为UOS/Deepin
    if os.path.exists("/etc/os-version"):
        with open("/etc/os-version", "r") as f:
            content = f.read()
            if "UnionTech" in content or "Deepin" in content:
                print("检测到统信UOS/深度系统")
                return True
    
    # 检查/etc/issue
    if os.path.exists("/etc/issue"):
        with open("/etc/issue", "r") as f:
            content = f.read()
            print("系统信息: " + content.strip())
    
    return True

def check_firefox_installed():
    """检查Firefox是否安装"""
    print("\n" + "="*60)
    print("检查Firefox安装状态")
    print("="*60)
    
    # 方法1: 检查firefox命令
    returncode, output, error = run_command(["which", "firefox"], False)
    if returncode == 0:
        print("✓ Firefox已安装，路径: " + output.strip())
        return True
    
    # 方法2: 检查/usr/bin/firefox
    if os.path.exists("/usr/bin/firefox"):
        print("✓ Firefox已安装，路径: /usr/bin/firefox")
        return True
    
    # 方法3: 检查包管理器
    returncode, output, error = run_command(["dpkg", "-l", "firefox"], False)
    if returncode == 0 and "firefox" in output:
        print("✓ Firefox已通过dpkg安装")
        return True
    
    print("✗ Firefox未安装")
    return False

def get_firefox_version():
    """获取Firefox版本"""
    print("\n获取Firefox版本信息...")
    
    methods = [
        ["firefox", "--version"],
        ["/usr/bin/firefox", "--version"],
        ["firefox-esr", "--version"]
    ]
    
    for cmd in methods:
        if cmd[0] != "firefox-esr" or os.path.exists("/usr/bin/firefox-esr"):
            returncode, output, error = run_command(cmd, False)
            if returncode == 0 and output:
                version_line = output.strip()
                if "Mozilla Firefox" in version_line:
                    version = version_line.split("Mozilla Firefox")[-1].strip()
                    print("当前版本: Mozilla Firefox " + version)
                    return version
    
    print("无法获取版本信息")
    return None

def update_firefox():
    """更新Firefox"""
    print("\n" + "="*60)
    print("开始更新Firefox")
    print("="*60)
    
    # 记录当前版本
    old_version = get_firefox_version()
    
    # 更新软件包列表
    print("\n1. 更新软件包列表...")
    run_command(["sudo", "apt", "update"])
    
    # 更新Firefox
    print("\n2. 更新Firefox...")
    run_command(["sudo", "apt", "install", "--only-upgrade", "firefox", "-y"])
    
    # 清理
    print("\n3. 清理缓存...")
    run_command(["sudo", "apt", "autoremove", "-y"])
    run_command(["sudo", "apt", "clean"])
    
    # 检查更新结果
    print("\n4. 验证更新结果...")
    new_version = get_firefox_version()
    
    if old_version and new_version:
        if old_version != new_version:
            print("\n✓ 更新成功！")
            print("  更新前: " + old_version)
            print("  更新后: " + new_version)
        else:
            print("\n✓ Firefox已经是最新版本")
    else:
        print("\n⚠ 无法验证更新结果，请手动检查")

def install_firefox():
    """安装Firefox"""
    print("\n" + "="*60)
    print("安装Firefox")
    print("="*60)
    
    print("Firefox未安装，现在开始安装...")
    
    # 更新软件包列表
    run_command(["sudo", "apt", "update"])
    
    # 安装Firefox
    run_command(["sudo", "apt", "install", "firefox", "-y"])
    
    # 验证安装
    if check_firefox_installed():
        get_firefox_version()
        print("\n✓ Firefox安装完成！")
    else:
        print("\n✗ Firefox安装失败")

def main_menu():
    """显示主菜单"""
    print("\n" + "="*60)
    print("统信UOS Firefox 管理工具")
    print("="*60)
    
    if check_firefox_installed():
        print("\n请选择操作:")
        print("  1. 更新Firefox")
        print("  2. 查看当前版本")
        print("  3. 重新安装Firefox")
        print("  4. 卸载Firefox")
        print("  0. 退出")
        
        choice = input("\n请输入选择 (0-4): ").strip()
        
        if choice == "1":
            update_firefox()
        elif choice == "2":
            get_firefox_version()
        elif choice == "3":
            run_command(["sudo", "apt", "install", "--reinstall", "firefox", "-y"])
        elif choice == "4":
            run_command(["sudo", "apt", "remove", "firefox", "-y"])
            print("Firefox已卸载")
        elif choice == "0":
            print("退出程序")
            return False
        else:
            print("无效选择")
    else:
        print("\nFirefox未安装")
        choice = input("是否安装Firefox？ (y/n): ").strip().lower()
        if choice == "y" or choice == "yes":
            install_firefox()
        else:
            print("取消安装")
    
    return True

def main():
    """主函数"""
    # 检查系统
    if not check_system():
        return
    
    # 检查权限
    if os.geteuid() != 0:
        print("\n注意: 部分操作需要管理员权限")
        print("请确保您有sudo权限")
    
    # 显示菜单
    while main_menu():
        input("\n按回车键继续...")
    
    print("\n感谢使用！")

if __name__ == "__main__":
    main()
```
