import subprocess

def run_in_terminal(command):
    """在新终端中运行命令"""
    # 检测默认终端
    default_terminal = "deepin-terminal"
    
    try:
        # 尝试打开终端并执行命令
        if default_terminal == "deepin-terminal":
            # UOS深度终端
            subprocess.Popen([default_terminal, "-e", "bash -c '{}; exec bash'".format(command)])
        elif default_terminal == "gnome-terminal":
            # GNOME终端
            subprocess.Popen(["gnome-terminal", "--", "bash", "-c", command])
        else:
            # 通用方法
            subprocess.Popen([default_terminal, "-e", command])
        
        print(f"已在{default_terminal}中执行: {command}")
        return True
    except Exception as e:
        print(f"错误: {e}")
        # 尝试直接执行命令
        print("直接执行命令:")
        result = subprocess.run(command, shell=True, text=True)
        print(result.stdout)
        return False
