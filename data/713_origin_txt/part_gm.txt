```bash

  -f ~/.ssh/id_ed25519 -N "" -q

chmod 644 ~/.ssh/id_ed25519.pub

Host github.com
    HostName github.com
    User git
    IdentityFile ~/.ssh/id_ed25519
    IdentitiesOnly yes

chmod 600 ~/.ssh/config
```
