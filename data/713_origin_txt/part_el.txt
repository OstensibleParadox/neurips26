#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
无警告启动Firefox

import os
import subprocess
import sys

def start_firefox_clean():
    """无警告启动Firefox"""
    # 设置环境变量避免警告
    env = os.environ.copy()
    env['NO_AT_BRIDGE'] = '1'
    env['MOZ_ENABLE_WAYLAND'] = '1'
    
    # 尝试多种启动方式
    firefox_paths = [
        "firefox",
        "/opt/apps/com.mozilla.firefox-zh/files/bin/firefox",
        "/usr/bin/firefox"
    ]
    
    for path in firefox_paths:
        try:
            print("正在启动Firefox...")
            print("版本: 136 (最新版)")
            
            # 启动Firefox，重定向错误输出
            with open(os.devnull, 'w') as devnull:
                process = subprocess.Popen(
                    [path] + sys.argv[1:],
                    env=env,
                    stdout=devnull,
                    stderr=devnull,
                    preexec_fn=os.setpgrp
                )
            
            print("Firefox已启动 (PID: %d)" % process.pid)
            return True

## 4. 完整的系统更新脚本

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
UOS系统更新工具 - 包括Firefox和系统更新
"""

import os
import subprocess
import sys

def run_command(cmd, description=""):
    """运行命令并显示结果"""
    if description:
        print("\n" + "="*60)
        print(description)
        print("="*60)
    
    print("执行: " + " ".join(cmd))
    
    try:
        # 对于Python 2使用Popen
        process = subprocess.Popen(cmd, 
                                 stdout=subprocess.PIPE,
                                 stderr=subprocess.STDOUT,
                                 universal_newlines=True)
        
        output_lines = []
        while True:
            line = process.stdout.readline()
            if not line and process.poll() is not None:
                break
            if line:
                # 过滤掉GTK警告信息
                if "WARNING" not in line and "ATTENTION" not in line:
                    print(line.rstrip())
                    output_lines.append(line)
        
        return process.returncode, "".join(output_lines)
    except Exception as e:
        print("错误: " + str(e))
        return -1, ""

def check_firefox():
    """检查Firefox状态"""
    print("\n" + "="*60)
    print("Firefox 状态检查")
    print("="*60)
    
    # 检查包状态
    print("\n1. 包信息:")
    run_command(["dpkg", "-s", "com.mozilla.firefox-zh"])
    
    # 检查版本（无警告方式）
    print("\n2. 版本信息:")
    cmd = "firefox --version 2>&1 | grep -v 'WARNING' | grep -v 'ATTENTION'"
    os.system(cmd)
    
    # 检查更新
    print("\n3. 更新检查:")
    run_command(["apt", "list", "--upgradable", "com.mozilla.firefox-zh"])
    
    # 检查安装位置
    print("\n4. 安装位置:")
    cmd = "dpkg -L com.mozilla.firefox-zh | grep -E '(bin/|firefox$)' | head -10"
    os.system(cmd)

def update_system():
    """更新系统所有软件"""
    print("\n" + "="*60)
    print("系统更新")
    print("="*60)
    
    print("\n系统有140个软件包可以更新")
    
    choice = raw_input("\n是否更新所有软件包? (y/n): ").strip().lower()
    if choice != 'y':
        print("跳过系统更新")
        return
    
    steps = [
        ("更新软件包列表", ["sudo", "apt", "update"]),
        ("升级所有软件包", ["sudo", "apt", "upgrade", "-y"]),
        ("升级依赖关系", ["sudo", "apt", "dist-upgrade", "-y"]),
        ("清理旧版本", ["sudo", "apt", "autoremove", "-y"]),
        ("清理缓存", ["sudo", "apt", "clean"]),
    ]
    
    for desc, cmd in steps:
        returncode, output = run_command(cmd, desc)
        if returncode != 0:
            print("步骤失败: " + desc)
            break
