    def __init__(self, similarity_threshold=0.85, node_limit=None):

    def build(self, text):
        """构建叙事图"""
        
        print(f"\n📖 Processing text ({len(text):,} characters)...")
        
        sentences = self.split_sentences(text)

            print("❌ No sentences extracted!")
