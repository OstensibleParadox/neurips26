    def split_sentences(self, text):
        """将文本分割成句子"""
        
        # 移除代码块标记
        text = re.sub(r'```\w*\n', '', text)
        text = re.sub(r'\n```', '', text)
        
        # 统一换行符

        sentences = []
        
        # 首先按段落分割
        paragraphs = [p.strip() for p in text.split('\n\n') if p.strip()]

                continue
                
            # 中英文混合句子分割

            para_sentences = re.split(split_pattern, para)

                if sent and len(sent) >= 3:
                    # 清理
                    sent = re.sub(r'^[\s　\-—*#]+', '', sent)
                    sent = re.sub(r'[\s　\-—*#]+$', '', sent)
                    if sent:
                        sentences.append(sent)
        
        # 后备方案
        if len(sentences) < 5:
            lines = [line.strip() for line in text.split('\n') if line.strip()]
            valid_lines = []
            for line in lines:
                if len(line) >= 4 and not line.startswith(('#', '**', '//', '--')):
                    valid_lines.append(line)
            sentences = valid_lines
        
        # 应用节点限制
        if self.node_limit is not None and len(sentences) > self.node_limit:
            print(f"  Limiting to first {self.node_limit} nodes")
            return sentences[:self.node_limit]
        
        print(f"  Extracted {len(sentences)} narrative units")

    def extract_label(self, text):
        """为节点创建简短标签"""
        
        clean_text = text.replace('*', '').replace('_', '').replace('#', '').strip()
        
        if len(clean_text) <= 12:
            return clean_text
        
        # 提取中文
        chinese_chars = re.findall(r'[\u4e00-\u9fff]{2,}', clean_text)
        if chinese_chars:
            label_parts = []
            for chars in chinese_chars[:3]:
                if 2 <= len(chars) <= 6:
                    label_parts.append(chars)
            
            if label_parts:
                label = ''.join(label_parts)
                if len(label) > 10:
                    label = label[:8] + ".."
                return label
        
        # 提取英文
        english_words = re.findall(r'\b[A-Za-z]{3,}\b', clean_text)
        if english_words:
            meaningful_words = []
            stop_words = {'the', 'and', 'but', 'for', 'you', 'are', 'was', 'were', 'that', 'this', 'with'}
            
            for word in english_words:
                word_lower = word.lower()
                if word_lower not in stop_words and len(word) >= 3:
                    meaningful_words.append(word)
                    if len(meaningful_words) >= 2:
                        break

                label = ' '.join(meaningful_words[:2])
                if len(label) > 10:

                return label

            return clean_text[:8] + ".."
