### 1. 首先检查系统已安装的浏览器

### 2. 修改Python脚本，使用正确的包名

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
统信UOS Firefox/浏览器更新脚本
"""

import subprocess
import os
import sys

def check_browsers():
    """检查系统中安装的浏览器"""
    print("=" * 60)
    print("检查系统中已安装的浏览器")
    print("=" * 60)
    
    browsers = []
    
    # 检查firefox相关包
    cmd = "dpkg -l | grep -i firefox | awk '{print $2 \" \" $3}'"
    result = subprocess.run(cmd, shell=True, capture_output=True, text=True)
    if result.stdout.strip():
        print("Firefox相关包:")
        for line in result.stdout.strip().split('\n'):
            if line:
                browsers.append(("firefox_package", line))
                print("  " + line)
    
    # 检查chromium相关包
    cmd = "dpkg -l | grep -i chromium | awk '{print $2 \" \" $3}'"
    result = subprocess.run(cmd, shell=True, capture_output=True, text=True)
    if result.stdout.strip():
        print("\nChromium相关包:")
        for line in result.stdout.strip().split('\n'):
            if line:
                browsers.append(("chromium_package", line))
                print("  " + line)
    
    # 检查可执行文件
    browser_paths = [
        ("firefox", ["firefox", "firefox-esr"]),
        ("chromium", ["chromium", "chromium-browser", "google-chrome"]),
        ("其他", ["opera", "vivaldi", "brave-browser"])
    ]
    
    print("\n可执行浏览器:")
    for browser_type, cmds in browser_paths:
        for cmd in cmds:
            result = subprocess.run(["which", cmd], capture_output=True, text=True)
            if result.returncode == 0:
                browsers.append((browser_type, cmd + ": " + result.stdout.strip()))
                print("  %s: %s" % (browser_type, result.stdout.strip()))
    
    return browsers

def check_firefox_installation():
    """检查Firefox安装情况"""
    print("\n" + "=" * 60)
    print("Firefox详细检查")
    print("=" * 60)
    
    # 检查包名变体
    possible_packages = [
        "firefox",                # 标准包名
        "firefox-esr",           # 企业版
        "firefox-l10n-zh-cn",    # 中文语言包
        "mozilla-firefox",       # 变体
        "firefox-ublock-origin", # 扩展版
    ]
    
    installed = []
    for package in possible_packages:
        cmd = "dpkg -l " + package + " 2>/dev/null | grep ^ii"
        result = subprocess.run(cmd, shell=True, capture_output=True, text=True)
        if result.stdout.strip():
            parts = result.stdout.strip().split()
            if len(parts) >= 3:
                installed.append((parts[1], parts[2]))
    
    if installed:
        print("已安装的Firefox包:")
        for package, version in installed:
            print("  %s: %s" % (package, version))
    else:
        print("未发现Firefox包安装")
        
        # 检查应用商店安装
        print("\n检查应用商店安装...")
        if os.path.exists("/opt/apps/cn.firefox"):
            print("  发现应用商店Firefox: /opt/apps/cn.firefox")
        elif os.path.exists("/opt/apps/org.mozilla.firefox"):
            print("  发现应用商店Firefox: /opt/apps/org.mozilla.firefox")
    
    return installed

def install_firefox_from_repo():
    """从仓库安装Firefox"""
    print("\n" + "=" * 60)
    print("尝试安装Firefox")
    print("=" * 60)
    
    # 首先查看可用的firefox包
    print("1. 搜索可用的firefox包...")
    subprocess.run(["apt", "search", "firefox"])
    
    print("\n2. 查看软件源中的firefox包...")
    subprocess.run(["apt-cache", "policy", "firefox"])
    subprocess.run(["apt-cache", "policy", "firefox-esr"])
    
    print("\n3. 尝试安装firefox-esr（企业版）...")
    result = subprocess.run(["sudo", "apt", "install", "firefox-esr", "-y"], 
                          capture_output=True, text=True)
    
    if result.returncode == 0:
        print("✓ firefox-esr 安装成功")
        return True
    
    print("\n4. 尝试其他可能的方法...")
    print("注意: UOS可能通过应用商店提供Firefox")
    print("建议:")
    print("  a. 打开UOS应用商店")
    print("  b. 搜索 'firefox' 或 '火狐'")
    print("  c. 点击安装")
    
    return False

def install_firefox_from_website():
    """从官网下载安装"""
    print("\n" + "=" * 60)
    print("从Mozilla官网下载安装")
    print("=" * 60)
    
    # 创建安装目录
    install_dir = os.path.expanduser("~/firefox")
    if not os.path.exists(install_dir):
        os.makedirs(install_dir)
    
    print("1. 下载Firefox...")
    # 下载最新版Firefox（中文版）
    download_url = "https://download.mozilla.org/?product=firefox-latest-ssl&os=linux64&lang=zh-CN"
    
    cmd = "wget -O %s/firefox.tar.bz2 '%s'" % (install_dir, download_url)
    print("下载命令: " + cmd)
    
    choice = input("\n是否从官网下载Firefox? (y/n): ").lower()
    if choice == 'y':
        subprocess.run(cmd, shell=True)
        
        # 解压
        print("\n2. 解压文件...")
        os.chdir(install_dir)
        subprocess.run(["tar", "-xjf", "firefox.tar.bz2"])
        
        # 创建桌面快捷方式
        print("\n3. 创建快捷方式...")
        desktop_file = os.path.expanduser("~/Desktop/Firefox.desktop")
        with open(desktop_file, "w") as f:
            f.write("""[Desktop Entry]
Name=Firefox
Comment=火狐浏览器
Exec=%s/firefox/firefox
Icon=%s/firefox/browser/chrome/icons/default/default128.png
Terminal=false
Type=Application
Categories=Network;WebBrowser;
""" % (install_dir, install_dir))
        
        subprocess.run(["chmod", "+x", desktop_file])
        
        print("\n✓ Firefox 已安装到: " + install_dir + "/firefox")
        print("  桌面快捷方式已创建")
        return True
    
    return False

def update_system():
    """更新系统所有软件"""
    print("\n" + "=" * 60)
    print("系统更新")
    print("=" * 60)
    
    print("注意: 系统有140个包可更新")
    
    choice = input("\n是否更新所有可用更新? (y/n): ").lower()
    if choice == 'y':
        print("开始更新系统...")
        subprocess.run(["sudo", "apt", "update"])
        subprocess.run(["sudo", "apt", "upgrade", "-y"])
        subprocess.run(["sudo", "apt", "autoremove", "-y"])
        print("✓ 系统更新完成")
    else:
        print("跳过系统更新")

def main():
    """主函数"""
    print("统信UOS 浏览器管理工具")
    print("版本: 1.0")
    
    while True:
        print("\n" + "=" * 60)
        print("主菜单")
        print("=" * 60)
        print("1. 检查已安装的浏览器")
        print("2. 检查Firefox安装详情")
        print("3. 从仓库安装Firefox")
        print("4. 从官网下载安装Firefox")
        print("5. 更新系统所有软件")
        print("6. 退出")
        
        choice = input("\n请选择操作 (1-6): ").strip()
        
        if choice == "1":
            browsers = check_browsers()
            if not browsers:
                print("\n未发现浏览器安装，建议安装Firefox")
        elif choice == "2":
            installed = check_firefox_installation()
            if not installed:
                print("\n系统未安装Firefox")
        elif choice == "3":
            install_firefox_from_repo()
        elif choice == "4":
            install_firefox_from_website()
        elif choice == "5":
            update_system()
        elif choice == "6":
            print("退出程序")
            break
        else:
            print("无效选择")
        
        input("\n按回车键继续...")

if __name__ == "__main__":
    main()
```

## 如果UOS确实没有Firefox，建议：

1. **使用应用商店安装**：
   - 打开UOS应用商店
   - 搜索"火狐"或"Firefox"
   - 点击安装

2. **手动下载安装**：
   ```bash
   # 下载Firefox
   wget https://download.mozilla.org/?product=firefox-latest-ssl&os=linux64&lang=zh-CN -O firefox.tar.bz2
   
   # 解压
   tar -xjf firefox.tar.bz2
   
   # 运行
   ./firefox/firefox
   ```

3. **使用其他浏览器**：
   ```bash
   # 安装Chromium
   sudo apt install chromium -y
   
   # 或者安装Edge
   # 先下载Edge的deb包，然后安装
   ```

### 2. 修改Python脚本，使用正确的包名

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
统信UOS Firefox更新脚本
针对 com.mozilla.firefox-zh 包
"""

import subprocess
import os
import sys

def get_firefox_version():
    """获取Firefox版本"""
    print("正在检查Firefox版本...")
    
    # 尝试多种方法获取版本
    methods = [
        ["firefox", "--version"],
        ["/opt/apps/com.mozilla.firefox-zh/files/bin/firefox", "--version"],
        ["/usr/bin/firefox", "--version"],
    ]
    
    for cmd in methods:
        try:
            result = subprocess.run(cmd, 
                                  capture_output=True, 
                                  text=True,
                                  timeout=5)
            if result.returncode == 0:
                output = result.stdout.strip() or result.stderr.strip()
                if "Mozilla Firefox" in output:
                    version = output.split("Mozilla Firefox")[-1].strip()
                    print("当前版本: %s" % version)
                    return version
        except:
            continue
    
    print("无法通过命令行获取版本，尝试其他方法...")
    
    # 检查包管理器中的版本
    cmd = "dpkg -l com.mozilla.firefox-zh | grep ^ii | awk '{print $3}'"
    result = subprocess.run(cmd, shell=True, capture_output=True, text=True)
    if result.stdout.strip():
        version = result.stdout.strip()
        print("包管理器版本: %s" % version)
        return version
    
    return None

def check_firefox_package():
    """检查Firefox包状态"""
    print("\n" + "="*60)
    print("检查Firefox包状态")
    print("="*60)
    
    # 检查包信息
    packages = [
        "com.mozilla.firefox-zh",  # 中文版Firefox
