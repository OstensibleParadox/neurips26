```python
import re

def enhance_for_obsidian(input_file, output_file):
    with open(input_file, 'r', encoding='utf-8') as f:
        content = f.read()
    
    # 1. 转换标准链接为Obsidian内部链接（如果链接指向.md文件）
    content = re.sub(r'\[([^\]]+)\]\(([^)]+)\.md\)', r'[[\2.md|\1]]', content)
    
    # 2. 增强表格对齐（Obsidian支持）
    content = re.sub(r'\|:---\|', r'|:---|', content)
    content = re.sub(r'\|---:\|', r'|---:|', content)
    content = re.sub(r'\|:---:\|', r'|:---:|', content)
    
    # 3. 添加YAML frontmatter（Obsidian支持）
    frontmatter = """---
aliases: []
tags: []
---
"""
    content = frontmatter + content
    
    with open(output_file, 'w', encoding='utf-8') as f:
        f.write(content)

# 使用
enhance_for_obsidian('output.md', 'obsidian-ready.md')
```
