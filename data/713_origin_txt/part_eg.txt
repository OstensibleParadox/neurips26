```python
import winreg
import re

def get_firefox_version_from_registry():
    """从Windows注册表获取Firefox版本"""
    firefox_versions = []
    
    # Firefox注册表路径
    registry_paths = [
        r"SOFTWARE\Mozilla\Mozilla Firefox",
        r"SOFTWARE\Wow6432Node\Mozilla\Mozilla Firefox",
        r"SOFTWARE\Mozilla\Firefox",
    ]
    
    for reg_path in registry_paths:
        try:
            key = winreg.OpenKey(winreg.HKEY_LOCAL_MACHINE, reg_path)
            
            # 获取所有版本
            for i in range(0, winreg.QueryInfoKey(key)[0]):
                try:
                    subkey_name = winreg.EnumKey(key, i)
                    # 检查是否是版本号格式（如 128.0.2 或 128.0.2 (x64 zh-CN)）
                    if re.match(r'\d+\.\d+(\.\d+)?', subkey_name.split()[0]):
                        subkey = winreg.OpenKey(key, subkey_name)
                        
                        try:
                            # 获取安装路径
                            install_path = winreg.QueryValueEx(subkey, "PathToExe")[0]
                            # 获取主版本号
                            version = subkey_name.split()[0]
                            
                            firefox_versions.append({
                                "version": version,
                                "install_path": install_path,
                                "bitness": "x64" if "x64" in subkey_name else "x86"
                            })
                        except (FileNotFoundError, OSError):
                            continue
                        
                        winreg.CloseKey(subkey)
                except (OSError, WindowsError):
                    continue
            
            winreg.CloseKey(key)
        except (OSError, WindowsError):
            continue
    
    # 返回最新版本
    if firefox_versions:
        # 按版本号排序（降序）
        firefox_versions.sort(key=lambda x: [int(n) for n in x["version"].split('.')], reverse=True)
        return firefox_versions[0]["version"]
    
    return None

# 使用方法

if reg_version:
    print(f"从注册表获取的Firefox版本: {reg_version}")
```

```python
def get_firefox_version_package_manager():
    """通过包管理器获取Firefox版本"""
    package_managers = [
        # dpkg (Debian/Ubuntu/UOS)
        (["dpkg", "-l", "firefox*"], "dpkg"),
        # apt (显示已安装版本)
        (["apt", "list", "--installed", "firefox*"], "apt"),
        # snap
        (["snap", "list", "firefox"], "snap"),
        # flatpak
        (["flatpak", "list", "--app", "--columns=application,version"], "flatpak"),
    ]
    
    print("通过包管理器查询Firefox版本:")
    print("-" * 40)
    
    for cmd, manager_name in package_managers:
        try:
            result = subprocess.run(cmd, 
                                  capture_output=True, 
                                  text=True,
                                  timeout=5)
            
            if result.returncode == 0 and result.stdout:
                lines = result.stdout.strip().split('\n')
                
                for line in lines:
                    line_lower = line.lower()
                    if ("firefox" in line_lower and 
                        not "lib" in line_lower and  # 排除库文件
                        not "language" in line_lower):  # 排除语言包
                        
                        if manager_name == "dpkg":
                            # dpkg输出格式: ii  firefox    128.0.2-1~deb12u1  amd64
                            parts = line.split()
                            if len(parts) >= 3 and "firefox" in parts[1]:
                                version = parts[2]
                                print(f"{manager_name}: Firefox {version}")
                                return version
                        
                        elif manager_name == "apt":
                            # apt输出格式: firefox/现在 128.0.2-1~deb12u1 amd64 已安装
                            if "/" in line and "现在" in line or "installed" in line:
                                parts = line.split()
                                if "firefox" in parts[0]:
                                    version = parts[0].split('/')[1]
                                    print(f"{manager_name}: Firefox {version}")
                                    return version
                        
                        elif manager_name == "snap":
                            # snap输出格式: firefox  128.0.2  2024-xx-xx
                            parts = line.split()
                            if "firefox" == parts[0].lower():
                                print(f"{manager_name}: Firefox {parts[1]}")
                                return parts[1]
        
        except (subprocess.TimeoutExpired, FileNotFoundError):
            continue
    
    print("-" * 40)
    return None
```

```python
import os
import json
from pathlib import Path
import re

def get_firefox_version_from_config_uos():
    """从Firefox配置文件获取版本（UOS/Linux专用）"""
    config_paths = [
        # UOS/Deepin默认路径
        Path.home() / ".config/firefox",
        Path.home() / ".mozilla/firefox",
        # 全局安装路径
        Path("/etc/firefox"),
        Path("/usr/lib/firefox"),
    ]
    
    for config_path in config_paths:
        if config_path.exists():
            # 1. 检查profiles.ini
            profiles_ini = config_path / "profiles.ini"
            if profiles_ini.exists():
                try:
                    with open(profiles_ini, 'r', encoding='utf-8') as f:
                        content = f.read()
                    
                    # 查找默认配置路径
                    for line in content.split('\n'):
                        if line.startswith("Default=") or line.startswith("Path="):
                            profile_dir = line.split('=')[1].strip()
                            profile_path = config_path / profile_dir
                            
                            # 检查compatibility.ini
                            compat_file = profile_path / "compatibility.ini"
                            if compat_file.exists():
                                with open(compat_file, 'r', encoding='utf-8') as f:
                                    compat_content = f.read()
                                
                                # 查找版本信息
                                for compat_line in compat_content.split('\n'):
                                    if compat_line.startswith("LastVersion="):
                                        version = compat_line.split('=')[1].strip()
                                        return f"配置文件版本: {version}"
                except:
                    continue
            
            # 2. 检查distribution目录（企业部署版）
            distribution_file = config_path / "distribution" / "distribution.ini"
            if distribution_file.exists():
                try:
                    with open(distribution_file, 'r', encoding='utf-8') as f:
                        content = f.read()
                    
                    # 查找版本信息
                    match = re.search(r'version\s*=\s*([\d\.]+)', content, re.IGNORECASE)
                    if match:
                        return f"分发版版本: {match.group(1)}"
                except:
                    continue
    
    return None
```
