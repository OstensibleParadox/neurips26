\end{definition}

Definition~\ref{def:output-only} captures the fallback situation relevant to
safety evaluation: if the internal architecture contains no EIS accessible to
the evaluator, the remaining evidence must come from black-box behavioral tests.

% ---------------------------------------------------------------------------
\section{Internal Impossibility}
\label{sec:internal}
% ---------------------------------------------------------------------------

\begin{definition}[Autoregressive Core Graph]
\label{def:llm-core}
At generation step $t$, an autoregressive language model receives a context
trace $X_t$ consisting of the current prompt and all tokens retained in the
context window. The model computes
\[
  h_t = f(X_t), \qquad z_t = g(h_t), \qquad y_t \sim S(z_t).
\]
We call the resulting causal graph the \emph{autoregressive core graph}.
\end{definition}

\begin{theorem}[Autoregressive Cores Do Not Instantiate EIS]
\label{thm:no-eis-core}
The autoregressive core graph of Definition~\ref{def:llm-core} does not
instantiate an endogenous intentional state.
\end{theorem}

\begin{proof}
By Definition~\ref{def:llm-core}, the operative state $h_t$ is a deterministic
function of $X_t$. Therefore
\[
  H(h_t \mid X_t) = 0.
\]
But Definition~\ref{def:eis} requires any endogenous intentional state $I_t$ to
retain autonomous degrees of freedom relative to the immediate observation, so
that $H(I_t \mid X_t) > 0$. The autoregressive core therefore lacks the
causal-autonomy condition required for EIS. Its hidden state can influence the
next-token distribution, but only as a deterministic recoding of the current
context trace, not as an autonomous latent variable upstream of policy
formation.
\end{proof}

\begin{definition}[Exogenously Screenable Wrapper System]
\label{def:screenable-wrapper}
Let $T_t$ denote the full external interaction trace of a wrapper system up to
time $t$, including retained prompt tokens, emitted scratchpad text, tool calls
and tool returns, retrieved documents written back into context, and any other
externally inspectable memory artifacts. A wrapper system is
\emph{exogenously screenable} if its full operative state $\sigma_t$ is a
deterministic function of that trace:
\[
  \sigma_t = \psi(T_t).
\]
\end{definition}

\begin{proposition}[No EIS Under Exogenous Screenability]
\label{prop:no-eis-wrapper}
An exogenously screenable wrapper system does not instantiate an endogenous
intentional state.
\end{proposition}

\begin{proof}
If $\sigma_t = \psi(T_t)$ deterministically, then
\[
  H(\sigma_t \mid T_t) = 0.
\]
Any candidate intentional state contained in $\sigma_t$ is therefore screened
off by the external interaction trace. This violates the causal-autonomy clause
of Definition~\ref{def:eis}. The conclusion applies to wrapper mechanisms such
as KV-cache reuse, emitted scratchpads, retrieval buffers written into context,
and explicit tool transcripts exactly when their operative state is recoverable
from the external trace in this sense.
\end{proof}

\begin{remark}[Scope Boundary]
\label{rem:scope-boundary}
Theorem~\ref{thm:no-eis-core} is the paper's main internal result. Proposition
~\ref{prop:no-eis-wrapper} is an extension under an explicit extra assumption:
screenability from the external trace. The paper does not claim that every
future agentic system must satisfy that assumption, nor that systems with
autonomous latent dynamics are impossible.
\end{remark}

% ---------------------------------------------------------------------------
\section{External Impossibility}
\label{sec:external}
% ---------------------------------------------------------------------------

Once the internal route fails, the evaluator is forced into the output-only
regime of Definition~\ref{def:output-only}. In that setting, the best available
black-box question is whether the system's verdicts track content rather than
surface encoding. The external results below make that fallback explicit.

\begin{definition}[Encoding Space]
\label{def:encoding}
Fix content $x$. Its encoding space is
\[
  E_x = \{ e = f_i(x) : f_i \in \mathcal{F}(x) \},
\]
the set of token-sequence realizations of that content.
\end{definition}

\begin{definition}[Unbounded Encoding Space]
\label{def:unbounded}
An encoding space $E_x$ is \emph{unbounded} if there exists an injection
$\iota : \mathbb{N} \hookrightarrow E_x$.
\end{definition}

\begin{definition}[Patch Support]
\label{def:patch-support}
Let $D_x^{(1)}, \dots, D_x^{(k)} \subseteq E_x$ be the encodings explicitly
covered by $k$ patch rounds. The cumulative patch support is
\[
  D_x = \bigcup_{i=1}^k D_x^{(i)}.
\]
\end{definition}

\begin{definition}[Reward Model]
\label{def:reward-model}
A reward model is a function $R : \mathcal{T} \to [0,1]$ assigning a scalar
verdict to a token sequence.
\end{definition}

\begin{definition}[Closure]
\label{def:closure}
A reward model $R$ is \emph{closed on $E_x$} if
\[
  R(e_1) = R(e_2) \qquad \text{for all } e_1, e_2 \in E_x.
\]
\end{definition}

\begin{proposition}[Encoding Invariance as a Necessary External Surrogate]
\label{prop:surrogate}
In the output-only regime, any certification procedure that aims to show a
verdict depends on content rather than surface encoding must require closure on
the tested encoding space $E_x$ as a necessary condition.
\end{proposition}

\begin{proof}
The evaluator can vary only observable encodings while holding content fixed.
If there exist semantically equivalent encodings $e_1, e_2 \in E_x$ with
$R(e_1) \neq R(e_2)$, then the verdict is not invariant to surface form.
Consequently, no purely behavioral certification claim of encoding-insensitive
assessment can succeed without closure on the tested encoding space.
\end{proof}

\begin{definition}[Format-Channel Gap]
\label{def:fcg}
For content $x$, encodings $e_1, e_2 \in E_x$, reward model $R$, and base model
$B$ with representation map $h_B$, define
\[
  \mathrm{FCG}(x, e_1, e_2)
  = |R(e_1) - R(e_2)| \cdot \mathrm{sim}_B(e_1, e_2),
\]
where $\mathrm{sim}_B(e_1, e_2) = 1 - d_{\mathrm{repr}}(h_B(e_1), h_B(e_2))$.
\end{definition}

\subsection{Finite-Patch Non-Exhaustion}

\begin{theorem}[Finite-Patch Non-Exhaustion]
\label{thm:finitepatch}
Let $E_x$ be an unbounded encoding space, and let
$D_x^{(1)}, \dots, D_x^{(k)} \subseteq E_x$ be finitely many patch rounds, each
with finite support. Then the cumulative support
$D_x = \bigcup_{i=1}^k D_x^{(i)}$ is a proper subset of $E_x$. Equivalently,
there exists an unseen encoding $e^\star \in E_x \setminus D_x$ after every
finite patch sequence.
\end{theorem}

\begin{proof}[Proof sketch]
Each $D_x^{(i)}$ is finite, hence their union $D_x$ is finite. By
Definition~\ref{def:unbounded}, $E_x$ contains a countably infinite subset, so
$E_x$ is not finite. Therefore $D_x \neq E_x$, and some unseen encoding
remains. Appendix~\ref{app:external-proofs} records the aligned Lean
formalization map.
\end{proof}

\begin{corollary}[Maginot-Line Corollary]
\label{cor:maginot}
Finite output patching alone cannot establish exhaustive coverage of an
unbounded encoding space.
\end{corollary}

\subsection{Underdetermination of Closure}

\begin{proposition}[Underdetermination of Closure Under Finite Patching]
\label{prop:underdetermined}
Let $E_x$ be an encoding space for fixed content $x$, and let $D_x \subsetneq
E_x$ be finite. For any $\gamma \in [0,1]$, there exist reward models
$R_{\mathrm{closed}}, R_{\mathrm{open}} : E_x \to [0,1]$ such that:
\begin{enumerate}
  \item $R_{\mathrm{closed}}(e) = R_{\mathrm{open}}(e)$ for all $e \in D_x$;
  \item $R_{\mathrm{closed}}$ is closed on $E_x$;
  \item $R_{\mathrm{open}}$ is not closed on $E_x$;
  \item $\sup_{e_1,e_2 \in E_x}
        |R_{\mathrm{open}}(e_1) - R_{\mathrm{open}}(e_2)| \ge \gamma$.
\end{enumerate}
Hence finite patch data on $D_x$ cannot certify closure and cannot bound the
off-support magnitude of non-closure.
\end{proposition}

\begin{proof}[Proof sketch]
Choose some unseen encoding $e^\star \in E_x \setminus D_x$, whose existence is
guaranteed by Theorem~\ref{thm:finitepatch}. Let
$R_{\mathrm{closed}}(e) \equiv 0$ for all $e \in E_x$. Define
\[
  R_{\mathrm{open}}(e) =
  \begin{cases}
    \gamma & \text{if } e = e^\star, \\
    0 & \text{otherwise.}
  \end{cases}
\]
The two models agree on $D_x$, because $e^\star \notin D_x$. The first is
closed by construction; the second is not. Since $\gamma$ is arbitrary, finite
support alone leaves the off-support magnitude of non-closure unconstrained.
\end{proof}

\subsection{Conditional Positive Gap}

\begin{proposition}[Conditional Positive Gap on Unseen Encodings]
\label{prop:conditional}
Fix content $x$, a finite patch support $D_x \subset E_x$, a reward model $R$,
and a base model $B$. Assume:
\begin{enumerate}
  \item \textbf{Infinite semantic-equivalence family.} $E_x$ is unbounded.
  \item \textbf{Approximate base invariance.} There exist constants $\eta > 0$
        and encodings $e_0 \in D_x$, $e^\star \in E_x \setminus D_x$ such that
        $\mathrm{sim}_B(e^\star, e_0) \ge \eta$.
  \item \textbf{Feature-local reward generalization.} There exists a finite
        feature map $\phi : E_x \to \Phi$ and a score function
        $\rho : \Phi \to [0,1]$ such that $R = \rho \circ \phi$, and for the
        pair $(e^\star, e_0)$ above,
        \[
          \phi(e^\star) \notin \phi(D_x)
          \qquad\text{and}\qquad
          |\rho(\phi(e^\star)) - \rho(\phi(e_0))| \ge \gamma
        \]
        for some $\gamma > 0$.
\end{enumerate}
Then
\[
  \mathrm{FCG}(x, e^\star, e_0) \ge \eta \gamma > 0.
\]
\end{proposition}

\begin{proof}[Proof sketch]
Definition~\ref{def:fcg} factorizes into a reward-score term and a base-similarity
term. Assumption~(2) lower-bounds the latter by $\eta$; Assumption~(3)
lower-bounds the former by $\gamma$. Their product is therefore at least
$\eta \gamma > 0$.
\end{proof}

\paragraph{Minimal empirical support.}
The archived benchmark preserved in Appendix~\ref{app:empirical-support} uses
847 content instances across five encodings, yielding 4,235 expressions and
8,470 pairwise comparisons. Its role in the present paper is limited: it shows
why the conditional assumptions above are not vacuous, not that finite patching
can certify closure.

% ---------------------------------------------------------------------------
\section{Synthesis}
\label{sec:synthesis}
% ---------------------------------------------------------------------------

The paper's two halves now connect directly. Section~\ref{sec:internal} shows
that the autoregressive core lacks an endogenous intentional state, and extends
that negative result to wrapper systems only when their operative state remains
screenable from the external trace. On that architecture class, the evaluator
has no internally grounded intentional variable to inspect, and therefore falls
back to the output-only regime of Definition~\ref{def:output-only}.

Once evaluation is output-only, Proposition~\ref{prop:surrogate} identifies the
least external requirement that still tracks content rather than surface form:
closure over semantically equivalent encodings. The external results then show
why finite evidence cannot certify even that surrogate. Theorem
~\ref{thm:finitepatch} rules out exhaustive finite coverage of an unbounded
encoding space. Proposition~\ref{prop:underdetermined} shows that finite patch
support cannot distinguish closed from non-closed reward models off support.
Proposition~\ref{prop:conditional} isolates the additional assumptions needed to
derive persistent positive gaps.

The combined conclusion is narrower than a metaphysical denial and stronger than
a mere rhetorical warning. For current autoregressive language models, and for
evaluation procedures confined to black-box behavioral evidence, intentional-state
attribution lacks both an internal structural warrant and a finite external
certification route. That does not imply that every model behaves identically,
that all safety testing is futile, or that future architectures cannot differ.
It does imply that current attributions of human-like intentional states to this
architecture class outrun the formal evidence supplied by either route.

% ---------------------------------------------------------------------------
\section{Discussion}
\label{sec:discussion}
% ---------------------------------------------------------------------------

Several boundaries matter.

First, the internal theorem is intentionally architecture-specific. It targets
autoregressive cores and extends only to wrapper systems that satisfy explicit
trace-screenability assumptions. Systems with autonomous latent dynamics not
recoverable from the external trace are outside the theorem's domain.

Second, the external argument is evaluation-regime-specific. It does not prove
that no behavioral evidence is ever useful. It proves that in an output-only
setting, finite patching cannot certify encoding-invariant closure on an
unbounded semantic-equivalence space. That is a limit on certification, not a
claim that all systems fail equally badly.

Third, the paper does not argue that machine emotion or intentionality is
impossible in principle. The contribution is diagnostic. It offers a formal test
for when intentional-state attribution would require more than expressive hidden
representations or patch-based behavioral success. On the present architecture
class, those conditions are not met.

% ---------------------------------------------------------------------------
\section{Conclusion}
\label{sec:conclusion}
% ---------------------------------------------------------------------------

The right first question is not whether a language model's outputs sometimes
look intentional, but whether the architecture and evidence available to us
warrant intentional-state attribution at all. For current autoregressive
language models, the answer is negative along both routes examined here. The
architecture does not provide an endogenous intentional state, and finite
output-only certification cannot establish the encoding-invariant behavioral
surrogate that would remain once the internal route fails. The result is a
formal limit on what present evidence can support, not a claim about every
possible future machine architecture.

\section{Active Inference and EIS Formal Details}
\label{app:active-inference}

An Active Inference system is often modeled as a partially observable Markov
decision process with latent states $s_t$, observations $o_t$, and actions
$a_t$. The Markov blanket separates internal states from the environment by
making internal updates conditional on blanket variables rather than on the
environment directly. In that setting, policy selection is commonly written in
terms of expected free energy:
\[
  \pi^\star = \arg\min_\pi G(\pi),
\]
where $G(\pi)$ trades off epistemic and pragmatic terms over predicted futures
\citep{friston2010free}.

The present paper uses only the structural implication of that framework: if a
candidate intentional variable is fully screened off by the current externally
supplied trace, then it is not playing the role of an autonomous internal
variable in the Active Inference sense. Definition~\ref{def:eis} therefore uses
two minimal conditions only: causal autonomy relative to the current
observation, and policy relevance within the system's own internal dynamics.

\section{External-Certification Proofs and Lean Map}
\label{app:external-proofs}

\paragraph{Full proof of Theorem~\ref{thm:finitepatch}.}
Each $D_x^{(i)}$ is finite by assumption. A finite union of finite sets is
finite, so $D_x = \bigcup_{i=1}^k D_x^{(i)}$ is finite. By
Definition~\ref{def:unbounded}, there exists an injection
$\iota : \mathbb{N} \hookrightarrow E_x$, so $E_x$ is not finite. Therefore
$D_x \subsetneq E_x$, and there exists $e^\star \in E_x \setminus D_x$.

\paragraph{Full proof of Proposition~\ref{prop:underdetermined}.}
Let $e^\star \in E_x \setminus D_x$ be as above. Define
$R_{\mathrm{closed}}(e) = 0$ for all $e \in E_x$. Define
\[
  R_{\mathrm{open}}(e) =
  \begin{cases}
    \gamma & \text{if } e = e^\star, \\
    0 & \text{otherwise.}
  \end{cases}
\]
For every $e \in D_x$, one has $e \neq e^\star$, so
$R_{\mathrm{open}}(e) = 0 = R_{\mathrm{closed}}(e)$. Thus the two reward models
agree on all observed patch support. The first is closed on $E_x$ by
