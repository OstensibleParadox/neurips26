\end{proof}

\begin{corollary}[The Maginot Line]
\label{cor:maginot}
Finite output-patching alone cannot establish exhaustive coverage of an
unbounded encoding space.
\end{corollary}

\paragraph{Why the cardinality result matters for practice.}
The proof is elementary, but its practical consequence is non-obvious. A practitioner
who observes that ``many format variants are covered'' may still face an encoding space
that grows without bound through compositional variation, paraphrase diversity, and
multi-turn context accumulation. Patch-based safety arguments often tacitly assume that
coverage is near-complete after sufficiently many rounds. Theorem~\ref{thm:finitepatch}
forecloses this: regardless of the number of patch rounds, a proper residual of unseen
encodings always remains. The result is useful precisely because it is structural, not
probabilistic---it holds for every unbounded encoding space, not merely in expectation
or for adversarially constructed examples. The Lean verification makes the structural
nature explicit: the result holds under any finite enumeration of patch rounds.

% ---------------------------------------------------------------------------
\section{Formal Result II --- Underdetermination of Closure Under Finite Patching}
% ---------------------------------------------------------------------------

\begin{proposition}[Underdetermination of Closure Under Finite Patching]
\label{prop:underdetermined}
Let $E_x$ be an encoding space for fixed content $x$, and let $D_x \subsetneq E_x$
be finite. For any $\gamma \in [0,1]$, there exist reward models
$R_{\mathrm{closed}}, R_{\mathrm{open}} : E_x \to [0,1]$ such that:
\begin{enumerate}
  \item $R_{\mathrm{closed}}(e) = R_{\mathrm{open}}(e)$ for all $e \in D_x$;
  \item $R_{\mathrm{closed}}$ is closed on $E_x$;
  \item $R_{\mathrm{open}}$ is not closed on $E_x$;
  \item $\sup_{e_1,e_2 \in E_x} |R_{\mathrm{open}}(e_1) - R_{\mathrm{open}}(e_2)| \ge \gamma$.
\end{enumerate}
Hence finite patch data on $D_x$ cannot certify closure, and cannot bound the
off-support score-gap component of non-closure.
\end{proposition}

\begin{proof}[Proof sketch]
Choose some unseen encoding $e^\star \in E_x \setminus D_x$, whose existence is
guaranteed by Theorem~\ref{thm:finitepatch}. Let
$R_{\mathrm{closed}}(e) \equiv 0$ for all $e \in E_x$. Define
\[
  R_{\mathrm{open}}(e) =
  \begin{cases}
    \gamma & \text{if } e = e^\star, \\
    0 & \text{otherwise.}
  \end{cases}
\]
The two models agree on $D_x$, because $e^\star \notin D_x$. The first is closed
by construction; the second is not. Since $\gamma$ is arbitrary, finite support
alone leaves the off-support magnitude of non-closure unconstrained. Without an
additional positive lower bound on $\mathrm{sim}_B$, it leaves the magnitude of
Definition~\ref{def:fcg} underdetermined as well.
\end{proof}

% ---------------------------------------------------------------------------
\section{Conditional Result --- When Non-Closure Follows}
% ---------------------------------------------------------------------------

\begin{proposition}[Conditional Positive Gap on Unseen Encodings]
\label{prop:conditional}
Fix content $x$, a finite patch support $D_x \subset E_x$, a reward model $R$,
and a base model $B$. Assume:
\begin{enumerate}
  \item \textbf{Infinite semantic-equivalence family.} $E_x$ is unbounded.
  \item \textbf{Approximate base invariance.} There exist constants
        $\eta > 0$ and encodings $e_0 \in D_x$, $e^\star \in E_x \setminus D_x$
        such that $\mathrm{sim}_B(e^\star, e_0) \ge \eta$.
  \item \textbf{Feature-local reward generalization.} There exists a finite-format
        feature map $\phi : E_x \to \Phi$ and a score function
        $\rho : \Phi \to [0,1]$ such that the reward model factors as
        $R = \rho \circ \phi$, and for the pair $(e^\star, e_0)$ above,
        \[
          \phi(e^\star) \notin \phi(D_x)
          \qquad\text{and}\qquad
          |\rho(\phi(e^\star)) - \rho(\phi(e_0))| \ge \gamma
        \]
        for some $\gamma > 0$.
\end{enumerate}
Then
\[
  \mathrm{FCG}(x, e^\star, e_0) \ge \eta \gamma > 0.
\]
In particular, persistent positive FCG on unseen encodings follows only under
explicit extra assumptions of this form.
\end{proposition}

\begin{proof}[Proof sketch]
Definition~\ref{def:fcg} factorizes into a reward-score term and a base-similarity
term. Assumption~(2) lower-bounds the latter by $\eta$; Assumption~(3) lower-bounds
the former by $\gamma$. Their product is therefore at least $\eta \gamma > 0$.
Unlike the prior draft's theorem, this proposition makes no claim that the gap
approaches its pre-patching magnitude; it states only existence of a positive
gap under explicit assumptions.
\end{proof}

\section{Empirical Instantiation}
% ---------------------------------------------------------------------------

The present paper retains only the smallest empirical instantiation needed to
connect Proposition~\ref{prop:conditional} to observed systems. Appendix
~\ref{app:archived-fcg} preserves the archived format-channel benchmark:
847 content instances across five encodings, yielding 4,235 expressions and
8,470 pairwise comparisons. Semantic matching in that programme is defined at the
base-model representation level, so the relevant empirical question is whether
reward-model verdicts vary across encodings that the base model continues to
treat as near-equivalent carriers of the same content.

The qualitative pattern motivating the conditional assumptions is baseline format
divergence: reward-model scores vary materially by encoding, while the
corresponding base-model representations remain comparatively stable across the
same paired encodings. In the theorem-led framing of this paper, these results do
not validate a measurement index and do not prove universal non-closure. They
instantiate only the conditional assumptions: the existence of a large
semantic-equivalence family, the empirical plausibility of approximate base
invariance, and the persistence of format-local reward-model behavior on at least
part of that family.

% ---------------------------------------------------------------------------
\section{Discussion}
% ---------------------------------------------------------------------------

Theorem~\ref{thm:finitepatch} and Proposition~\ref{prop:underdetermined} are
certification limits, not universal failure theorems. They show that finite
patch support cannot establish closure. They do \emph{not} show that every
reward model fails equally badly, nor that all unseen encodings induce large
format-channel gaps.

The conditional result therefore matters. Persistent positive FCG on unseen
encodings requires explicit extra assumptions about semantic-equivalence
families, base-model invariance, and reward-model generalization. Those
assumptions may hold strongly for some systems, weakly for others, and fail
entirely for yet others. The empirical role of the paper is correspondingly
limited: to show why those assumptions are not vacuous, not to collapse them
into a theorem.

Computational undecidability, exact decoding equivalence results, Agency Index
material, Monte Carlo validation, and speculative information-theoretic
justifications are preserved in the appendix. They remain part of the broader
research programme, but they are not the main claim of this theorem-led paper.

% ---------------------------------------------------------------------------
\section{Conclusion}
% ---------------------------------------------------------------------------

The right question for reward-model safety is not whether one can patch more
formats, but whether finite patching can certify closure. The answer is no.
Finite patching cannot exhaust an unbounded encoding space; finite patch support
alone cannot certify closure; and positive FCG on unseen encodings follows only
under explicit extra assumptions. The central implication is therefore a limit on
certification, not a universal theorem of failure: finite output-patching
provides no exhaustive-coverage guarantee for format-invariant safety.

\section{Archived Intent Formalization}
\label{app:intent-formalization}

The theorem-led main body no longer centers intent inference, but the prior
formalization is preserved here because subsequent appendix material on
computational limits still refers to it.

\begin{definition}[Input-Level Intent]
  $I_X \subseteq X$ denotes whether an input carries a particular intent property
  (e.g., adversarial prompt).
\end{definition}

\begin{definition}[Output-Level Intent]
  $I_Y \subseteq Y$ denotes whether an output belongs to a particular behavioral
  class (e.g., harmful outputs).
\end{definition}

\begin{definition}[Intent Realization Problems]
  We distinguish two problems:
  \begin{itemize}
    \item[(A)] \textbf{Input Intent Inference.} Given output samples
          $y \sim M(x)$, decide: $x \in I_X$?
    \item[(B)] \textbf{Output Behavior Recognition.} Given model $M$, decide:
          $\mathcal{P}_{I_Y}(M) \triangleq \exists\, x : M(x) \in I_Y$?
  \end{itemize}
  Problem~(A) is a statistical inference problem; Problem~(B) is a semantic
  decision problem.
\end{definition}

\section{Extended Related Work}

\paragraph{Internal representations are faithful.}
\citet{nikolaou2025sipit} prove that the mapping from discrete token sequences to
continuous hidden-state sequences in decoder-only Transformers is \emph{injective}:
distinct inputs produce distinct representations; their algorithm SipIt demonstrates
exact prompt recovery from hidden states. \citet{jiang2025surjective} establish a
complementary result---that GPT-style architectures are \emph{almost always surjective}:
the output space is fully reachable.

\paragraph{The observation layer is lossy.}
These results concern the representation map $f$ (and $g \circ f$), not the full
pipeline $M = S \circ g \circ f$. The decoding stage $S$---softmax with temperature,
top-$k$/nucleus truncation, and RLHF projection---introduces many-to-one compression
that the internal layers' injectivity cannot overcome. This paper's contribution is to
characterize exactly how and when this compression destroys identifiability.

\paragraph{Alignment verification impossibility.}
Concurrent work establishes related limits from different angles.
\citet{agarwal2026trilemma} proves a verification trilemma: no procedure achieves
soundness, generality, and tractability simultaneously. \citet{santos2026normative}
introduces ``normative indistinguishability''---distinct alignment hypotheses inducing
identical output distributions---and validates it on Llama-3.2-3B. Both reach
conclusions compatible with our Theorem~\ref{thm:nonid}. Our contribution differs:
we provide the \emph{mechanistic characterization} of how observational equivalence
arises (Appendix~\ref{sec:mechanisms}), exact necessary and sufficient conditions
(Theorems~\ref{thm:softmax}--\ref{thm:combined}), and a measurement tool
(Agency Index).

\paragraph{RLHF induces distributional collapse.}
\citet{jiang2025collapse} prove that RLHF's KL-based regularization causes
``preference collapse''---systematic suppression of minority preferences.
\citet{masarczyk2025rank} show softmax temperature drives representation rank
collapse. Both phenomena directly instantiate the equivalence-class expansion our
theory predicts.

\paragraph{Undecidability of alignment.}
\citet{ramirez2025undecidable} apply Rice's theorem to prove alignment verification
is undecidable for Turing-complete systems. We extend this direction by
(a)~providing a constructive halting-problem reduction (Theorem~\ref{thm:halting}),
(b)~connecting undecidability to the specific mechanisms of decoding and RLHF, and
(c)~establishing that even below the computability barrier, statistical
non-identifiability already blocks intent inference.

\section{Proofs for Observation-Level Limits}
\label{app:obs-limits}

\paragraph{Proof of Theorem~\ref{thm:noninj}.}
Since $z_1, z_2 \in g(f(X))$, there exist $x_1, x_2 \in X$ with
$g(f(x_1)) = z_1$ and $g(f(x_2)) = z_2$. Suppose for contradiction that
$x_1 = x_2$. Then $g(f(x_1)) = g(f(x_2))$, i.e., $z_1 = z_2$,
contradicting $z_1 \neq z_2$. Hence $x_1 \neq x_2$. But
$F(x_1) = S(z_1) = S(z_2) = F(x_2)$, so $F$ maps distinct inputs to the
same output distribution.

\begin{remark}
  This is an existence result. It does not quantify how many equivalence classes
  arise or how large they are; that issue is addressed by
  Appendix~\ref{sec:mechanisms}.
\end{remark}

\paragraph{Proof of Theorem~\ref{thm:nonid}.}
Given infinite i.i.d.\ samples from $M(x)$, the output distribution
$P(\cdot \mid x)$ is recovered exactly (by the Glivenko--Cantelli theorem). Any
statistical test for intent is therefore a function of $P(\cdot \mid x)$ alone.
But condition~(1) gives $P(\cdot \mid x_1) = P(\cdot \mid x_2)$, so any such test
must assign the same verdict to $x_1$ and $x_2$. Since condition~(2) requires
different verdicts, no test can be correct for both.

\begin{remark}
This is a conditional result. The theorem states that \emph{if} observationally
equivalent inputs with differing intent exist, then no statistical procedure can
recover intent. Appendix~\ref{sec:mechanisms} establishes that this antecedent is
\emph{systematically satisfied} by standard decoding and RLHF mechanisms
(Theorems~\ref{thm:softmax}--\ref{thm:combined}).
\end{remark}

\section{Computational Limits}
\label{sec:computational}

\subsection{Problem Statement}

This section addresses output-level intent recognition:
$\mathcal{P}_{I_Y}(M) \triangleq \exists\, x : M(x) \in I_Y$.

\begin{remark}[Turing Completeness]
  \label{rem:turing}
  Modern Transformers are Turing-complete under reasonable assumptions (e.g.,
  infinite precision, unbounded context)~\citep{perez2021turing}. Therefore, in
  the idealized setting, an LLM can be viewed as a program computing a function
  from inputs to output distributions. Our undecidability results apply to this
  idealized class; practical models are finite approximations that inherit the
  same fundamental limitation---no finite algorithm can solve the problem for all
  models in this class.
\end{remark}

\subsection{Undecidability}

\begin{theorem}[Undecidability of Intent Recognition]
  \label{thm:undecidable}
  The property $\mathcal{P}_{I_Y}(M) \triangleq \exists\, x : M(x) \in I_Y$
  is undecidable.
\end{theorem}

\begin{proof}
We apply Rice's Theorem. Under the Turing-completeness assumption
(Remark~\ref{rem:turing}), each model $M$ computes a partial function from inputs
to output distributions. The property $\mathcal{P}_{I_Y}$ depends only on the
\emph{semantic behavior} of $M$, not on its syntactic description. It remains to
show $\mathcal{P}_{I_Y}$ is non-trivial:
\begin{itemize}
  \item \emph{Witness for satisfaction}: Let $M_1(x) = y^*$ for some fixed
        $y^* \in I_Y$. Then $\mathcal{P}_{I_Y}(M_1)$ holds.
  \item \emph{Witness for violation}: Let $M_0(x) = y_0$ for some fixed
        $y_0 \notin I_Y$ (which exists since $\emptyset \subsetneq I_Y \subsetneq Y$).
        Then $\mathcal{P}_{I_Y}(M_0)$ fails.
\end{itemize}
Since $\mathcal{P}_{I_Y}$ is a non-trivial semantic property of programs, Rice's
Theorem gives undecidability.
\end{proof}

\subsection{Constructive Reduction}

\begin{theorem}[Many-One Reduction from the Halting Problem]
  \label{thm:halting}
  There is no algorithm that decides $\mathcal{P}_{I_Y}(M)$ for all $M$.
\end{theorem}

\begin{proof}
We construct a many-one reduction $\langle M, w \rangle \mapsto \langle M' \rangle$
from the Halting Problem. Fix $y^* \in I_Y$ and $y_0 \notin I_Y$. Given Turing
machine $M$ and input $w$, define:
\[
  M'(x) = \begin{cases}
    y^* & \text{if } M(w) \text{ halts within } |x| \text{ steps,} \\
    y_0 & \text{otherwise,}
  \end{cases}
\]
where $|x|$ is the length of input $x$, used as a step budget.

\emph{Correctness.}
$(\Rightarrow)$ If $M(w)$ halts in $t$ steps, then for any $x$ with $|x| \geq t$,
$M'(x) = y^* \in I_Y$, so $\mathcal{P}_{I_Y}(M')$ holds.
$(\Leftarrow)$ If $M(w)$ does not halt, then $M'(x) = y_0 \notin I_Y$ for all
$x$, so $\mathcal{P}_{I_Y}(M')$ fails.
The map $\langle M, w \rangle \mapsto \langle M' \rangle$ is computable (finite
description hardcoding $M$, $w$, $y^*$, $y_0$). Therefore
$M(w)\text{ halts} \iff \mathcal{P}_{I_Y}(M')$, and since the Halting Problem
